### Hijri Calendar On-Actions ###

# The Hijri date is shown to Muslim players by default, as it was before the calendar became a choice.
# ME_choose_calendar_decision can still switch to the Gregorian or the Twelver Hijri calendar afterwards.
on_game_start = {
	on_actions = {
		me_hijri_default_calendar
	}
}

# Safety net for saves made before the default existed
yearly_global_pulse = {
	on_actions = {
		me_hijri_default_calendar
	}
}

me_hijri_default_calendar = {
	effect = {
		# Only ever applied once, so a later choice of the Gregorian calendar is not undone
		if = {
			limit = {
				NOT = { exists = global_var:me_calendar_default_set }
			}
			set_global_variable = {
				name = me_calendar_default_set
				value = yes
			}
			set_global_variable = {
				name = me_calendar_hijri_var
				value = yes
			}
		}
	}
}
