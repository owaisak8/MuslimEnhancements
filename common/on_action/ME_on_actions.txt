ME_yearly_global_pulse = {
	effect = {
		# Islamic Holy Days
		trigger_event = { on_action = ME_ihd_yearly_pulse }
		
		# Historical character spawner
		if = {
			limit = {
				NOT = {
					has_game_rule = no_historical_characters
				}
			}
			me_spawn_historical_characters_effect = yes
		}

		# Nukkari Revolt
		if = {
			limit = {
				NOT = { exists = global_var:nukkari_revolt_ended }
			}
			nukkari_event_launcher_effect = yes
		}

		# Ha-Mimist Invasion
		if = {
			limit = {
				NOT = { exists = global_var:hamimist_invasion_ended }
			}
			hamimist_event_launcher_effect = yes
		}

		# Kaduist Invasion
		if = {
			limit = {
				NOT = { exists = global_var:kaduist_invasion_ended }
			}
			kaduist_event_launcher_effect = yes
		}

		# Alid Egyptian Revolts
		## Ibn as-Sufi
		if = {
			limit = {
				NOT = { exists = global_var:ibn_as_sufi_revolt_ended }
				current_date >= 867.1.1
				current_date < 868.1.1
			}
			title:k_egypt.holder = {
				trigger_event = {
					id = me_alid_egypt_revolt.0001 
					days = { 6 7 }
				}
			}
		}
		## Bugha al-Akbar
		if = {
			limit = {
				NOT = { exists = global_var:bugha_al_akbar_revolt_ended }
				current_date >= 868.1.1
				current_date < 869.1.1
			}
			title:k_egypt.holder = {
				trigger_event = {
					id = me_alid_egypt_revolt.0002
					days = { 90 260 }
				}
			}
		}
		## Bugha al-Asghar
		if = {
			limit = {
				NOT = { exists = global_var:bugha_al_asghar_revolt_ended }
				current_date >= 869.1.1
				current_date < 870.1.1
			}
			title:k_egypt.holder = {
				trigger_event = {
					id = me_alid_egypt_revolt.0003
					days = { 100 140 }
				}
			}
		}
		
		# al-Ghazali death event
		if = {
			limit = {
				current_date >= 1112.1.1
				current_date < 1113.1.1
			}
			every_living_character = {
				limit = {
					rite = { rite_has_doctrine = muhammad_succession_sunni_doctrine }
					OR = {
						is_ai = no
						highest_held_title_tier >= tier_kingdom 
						has_title = title:d_sunni
					}
				}
				trigger_event = {
					id = me_misc_religion_event.0001 
					days = { 30 100 }
				}
			}
		}

		# Turkmen culture emergence.
		if = {
			limit = {
				current_date < 1115.1.1
				NOT = { has_global_variable = turkmen_culture_emergence }
				current_date >= 1075.1.1
				culture:turkish = {
					any_culture_county = {
						any_county_province = {
							count >= 5
							OR = {
								geographical_region = ghw_region_anatolia
								geographical_region = ghw_region_mainland_greece
								geographical_region = custom_k_thessalonika
							}
						}
					}
				}
			}
			random = {
				chance = 10
				modifier = {
					add = 20 # 30% chance
					current_date >= 1085.1.1
				}
				modifier = {
					add = 30 # 60% chance
					current_date >= 1095.1.1
				}
				modifier = {
					add = 40 # 100% chance
					current_date >= 1105.1.1
				}
				set_global_variable = {
					name = turkmen_culture_emergence
					value = yes
				}
				trigger_event = {
					id = me_culture_emergence.0001
					days = { 60 300 }
				}
			}
		}
	}
}

	}

me_on_county_occupied = {
	effect = {
		# Acre-Ma'ab exchange related modifier 
		if = {
			limit = {
				scope:county = { this = title:c_acre }
				has_global_variable = me_tyre_county_exchange_flag
			}
			add_county_modifier = {
				modifier = me_acre_occupied_modifier
				years = 1
			}
			trigger_event = {
				id = me_misc_event.0007 
				days = 365
			}
		}
		
		# Nukkari Lootings
		if = {
			limit = {
				# Check that it's relevant so we don't fire it *all* the time
				scope:war = { using_cb = me_nukkari_revolt_war }
				NOT = { exists = scope:county.var:looted_by_nukkaris }
			}
			# Each County should only be looted once
			scope:county = {
				set_variable = {
					name = looted_by_nukkaris
					value = yes
				}
			}
			# Send a message (and gold!) to the Nukkari leader
			send_interface_message = {
				type = event_war_good
				title = nukkaris_sieged_county.t
				left_icon = scope:previous_controller
				right_icon = scope:county
				add_gold = {
					add = scope:county.prosperity_level
					multiply = 2
				}
				show_as_tooltip = {
					scope:county = {
						nukkaris_loot_county_effect = yes
					}
				}
			}
			# Send a message to the County's owner and his leige
			scope:previous_controller = {
				send_interface_message = {
					type = event_war_bad
					title = nukkaris_sieged_county.t
					left_icon = root
					right_icon = scope:county
					scope:county = {
						nukkaris_loot_county_effect = yes
					}
				}
				every_liege_or_above = { 
					send_interface_message = {
						type = event_war_bad
						title = nukkaris_sieged_county.t
						left_icon = root
						right_icon = scope:county
						show_as_tooltip = {
							scope:county = {
								nukkaris_loot_county_effect = yes
							}
						}
					}
				}
			}
		}
	}
}

ME_ramadan_global_pulse = {
	effect = {
		every_living_character = {
			limit = {
				rite = { rite_has_doctrine = doctrine_me_fasting_month_ramadan }
				OR = {
					is_ai = no
					highest_held_title_tier >= tier_kingdom
					any_held_title = { is_head_of_faith = yes }
				}
			}
			trigger_event = {
				on_action = ME_ramadan_events
				days = { 1 14 }
			}
		}
		every_living_character = {
			if = {
				limit = {
					is_landed = yes
					is_ai = yes
					is_at_war = no
					rite = { rite_has_doctrine = doctrine_me_fasting_month_ramadan }
					capital_county.faith ?= faith
					NOT = { has_trait = greedy }
					OR = {
						highest_held_title_tier >= tier_kingdom
						any_held_title = { is_head_of_faith = yes }
					}
					short_term_gold >= major_gold_value
				}
				capital_county = { add_county_modifier = me_iftar_sponsorship_substantial_modifier }
			}
			else_if = {
				limit = {
					is_landed = yes
					is_ai = yes
					is_at_war = no
					rite = { rite_has_doctrine = doctrine_me_fasting_month_ramadan }
					capital_county.faith ?= faith
					NOT = { has_trait = greedy }
					highest_held_title_tier >= tier_duchy
					short_term_gold >= major_gold_value
				}
				capital_county = { add_county_modifier = me_iftar_sponsorship_modest_modifier }
			}
		}
		every_player = {
			limit = {
				is_landed = yes
				rite = { rite_has_doctrine = doctrine_me_fasting_month_ramadan }
			}
			trigger_event = me_ihd_events.9000
		}
	}
}
