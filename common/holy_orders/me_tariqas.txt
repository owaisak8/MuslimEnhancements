# The Sufi orders, built on the 1.20 Monastic Order system (vanilla defines no Muslim monastic orders).
# Each order champions one tenet and is led by its Shaykh from a leased lodge.
#
# At game start they are created by me_set_up_tariqas_effect, which marks the patron with
# var:me_tariqa_to_found so that exactly the intended order is picked.
# Outside game start they can be founded by Sunni rulers once their historical founding date has passed.

# The "sober" school of al-Junayd of Baghdad
ho_me_junaydiyya = {
	type = monastic
	government = monastic_holy_order_government
	title = d_me_junaydiyya
	name = "holy_order_me_junaydiyya"
	tenet = tenet_inner_journey

	trigger = {
		trigger_if = {
			limit = { scope:founder = { has_variable = monastic_game_start_founder } }
			scope:founder = {
				has_variable = me_tariqa_to_found
				var:me_tariqa_to_found = flag:junaydiyya
			}
		}
		trigger_else = {
			scope:founder.religion = religion:islam_religion
			scope:founder.rite = { rite_has_doctrine = muhammad_succession_sunni_doctrine }
		}
	}

	weight = {
		value = 5
		if = {
			limit = { scope:founder = { has_variable = me_tariqa_to_found } }
			add = 10000
		}
	}
}

	type = monastic

			limit = { scope:founder = { has_variable = monastic_game_start_founder } }

			scope:province = { geographical_region = world_middle_east_persia }
		}

	}
}
