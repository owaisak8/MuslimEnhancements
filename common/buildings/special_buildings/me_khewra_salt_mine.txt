# Salt Mines of Khewra (b_jhelum, province 3435). Khewra itself lies in the map's Katas Raj barony, whose one
# special slot holds the temples; Jhelum is the nearest free barony on the same bank of the river, and
# modern Jhelum District contains Khewra.
# (00_special_mines.txt). Salt output (me_khewra_values.txt) tops every other salt source in the game,
# a fully built salt mine district included (530 at tier 10), with copper-mine-scale mining tax; needs timber for the galleries and pack
# animals to carry the rock salt to market.

	}

	trade_good_need = {
