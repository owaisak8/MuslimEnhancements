### Hijri Calendar Scripted Triggers ###

# Is it currently Ramadan (Hijri month 9)?
me_is_ramadan_trigger = {
	me_hijri_month_value >= 9
	me_hijri_month_value < 10
}

# Does the character already have a Muharram modifier?
me_has_muharram_modifier_trigger = {
	OR = {
		has_character_modifier = me_muharram_deep_mourning_modifier
		has_character_modifier = me_muharram_moderate_mourning_modifier
		has_character_modifier = me_muharram_tatbir_modifier
		has_character_modifier = me_muharram_skipped_modifier
	}
}
