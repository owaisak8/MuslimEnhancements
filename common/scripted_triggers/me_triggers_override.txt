# Is allowed to marry (current marriage/concubine status ignored)
# Vanilla 1.20 trigger with one ME change: Islam has no celibate religious life, so among Muslims the 'devoted' and
# 'order_member' traits do not bar marriage. Eunuchs are still barred.
can_marry_common_trigger = {
	is_alive = yes
	### ME Addition ###
	trigger_if = {
		limit = { religion = religion:islam_religion }
		NOR = {
			has_trait = eunuch_1
			has_trait = beardless_eunuch
		}
	}
	trigger_else = {
		NOT = { has_trait_with_flag = can_not_marry }
		# These two no longer carry the flag (see me_trait_override.txt)
		NOR = {
			has_trait = devoted
			has_trait = order_member
		}
	}
	### End of ME Addition ###
	NOT = { has_variable = can_not_marry }

	trigger_if = {
		limit = {
			exists = primary_title
		}
		NOR = {
			# Block Mercenaries from marrying or showing up in marriage interactions
			primary_title = {
				is_mercenary_company = yes
			}
			is_leading_faction_type = peasant_faction
			is_leading_faction_type = populist_faction
			government_has_flag = government_is_herder
		}
	}

	# Clergy cannot marry if their faith disallows it
	trigger_if = {
		limit = { is_clergy = yes }
		marriage_clergy_can_marry_trigger = yes
	}
	trigger_if = {
		limit = {
			exists = liege.involved_activity.var:tournament_marriage_prize
			this = liege.involved_activity.var:tournament_marriage_prize
		}
		exists = scope:contest_winner
	}
}

# Vanilla 1.20 trigger with one ME change: the "monks can never marry" rule does not apply to Muslims
marriage_clergy_can_marry_trigger = {
	#Monks can never marry
	### ME Addition ###
	trigger_if = {
		limit = {
			NOT = { religion = religion:islam_religion }
		}
		NOR = {
			has_trait = devoted
			has_trait = order_member
		}
	}
	### End of ME Addition ###
	OR = {
		rite = { rite_has_parameter = clergy_can_marry }
		AND = {
			is_ruler = yes
			top_participant_group:the_christian_church ?= { has_participant_group_parameter = clergy_can_marry }
			trigger_if = {
				limit = { is_ai = yes }
				saeculum_obscurum_clergy_would_marry_trigger = yes
			}
		}
	}
}

#### Scripted Rule Triggers ####
can_raid_trigger = {
	OR = {
		## ME Additions ##
		AND = {
			has_game_rule = me_clan_can_raid_yes
			government_has_flag = ME_clan_government_can_raid_rule
		}
		AND = {
			has_game_rule = me_clan_can_raid_struggle
			government_has_flag = ME_clan_government_can_raid_rule
			NOT = { any_character_struggle = { involvement = involved } }
		}

		# Government logic.
		government_has_flag = government_can_raid_rule
		## Admin - Frontier themes
		vassal_contract_has_flag = admin_theme_can_raid
		## Clan - Ghazi
		vassal_contract_has_flag = vassal_contract_allow_raiding
		# Faith Logic.
		faith = { has_doctrine_parameter = faith_can_raid }
		has_personal_tenet_flag = unlock_destruction_raid_intent_personal
		# Dynasty logic.
		dynasty ?= {
			OR = {
				can_raid_dynasty_modifiers_that_enable_raiding_overseas_list_trigger = yes
				can_raid_dynasty_modifiers_that_enable_raiding_overland_list_trigger = yes
			}
		}
		# House logic.
		house ?= {
			OR = {
				can_raid_house_modifiers_that_enable_raiding_overseas_list_trigger = yes
				can_raid_house_modifiers_that_enable_raiding_overland_list_trigger = yes
			}
		}
		# Cultural logic.
		## With restrictions.
		AND = {
			can_raid_cultural_parameter_requirements_for_non_tribals_trigger = yes
			culture = {
				OR = {
					has_cultural_parameter = culture_can_raid_at_sea_even_if_feudal
					has_cultural_parameter = culture_can_raid_over_land_even_if_feudal
				}
			}
		}
		## Without restrictions.
		culture = {
			OR = {
				has_cultural_parameter = culture_can_raid_at_sea_even_if_feudal_no_restrictions
				has_cultural_parameter = culture_can_raid_over_land_even_if_feudal_no_restrictions
			}
		}
		# Trait logics.
		## With restrictions.
		AND = {
			can_raid_trait_flag_standard_requirements_trigger = yes
			has_trait_with_flag = trait_enable_raiding_per_standard_restrictions
		}
		## Without restrictions.
		has_trait_with_flag = trait_enable_raiding_without_restrictions
		has_variable = enable_raiding_without_restrictions
	}
	# Some faiths prohibit this.
	OR = {
		NOR = {
			rite ?= { rite_has_parameter = holy_wars_forbidden }
			rite ?= { rite_has_parameter = tenet_peace_of_god_raid_ban }
			government_has_flag = government_is_landless_adventurer
		}
		# But not for Nomads or Wanua
		government_has_flag = government_is_nomadic
		government_has_flag = government_is_wanua
	}
	is_landed = yes
}

trait_is_criminal_in_rite_trigger = {
	OR = {
		#KINSLAYER
		AND = {
			OR = {
				$TRAIT$ = trait:kinslayer_3
				$TRAIT$ = trait:kinslayer_2
				$TRAIT$ = trait:kinslayer_1
			}
			$RITE$ = { rite_has_parameter = kinslaying_any_dynasty_member_crime }
		}
		AND = {
			OR = {
				$TRAIT$ = trait:kinslayer_3
				$TRAIT$ = trait:kinslayer_2
			}
			$RITE$ = { rite_has_parameter = kinslaying_extended_family_crime }
		}
		AND = {
			$TRAIT$ = trait:kinslayer_3
			$RITE$ = { rite_has_parameter = kinslaying_close_kin_crime }
		}
		#WITCH
		AND = {
			$TRAIT$ = trait:witch
			$RITE$ = { rite_has_parameter = witchcraft_illegal }
		}
		#CANNIBAL
		AND = {
			$TRAIT$ = trait:cannibal
			NOT = {
				$RITE$ = { rite_has_parameter = cannibalism_legal }
			}
		}
		#SODOMITE
		AND = {
			$TRAIT$ = trait:sodomite
			$RITE$ = { rite_has_parameter = homosexuality_illegal }
		}
		#DEVIANT
		AND = {
			$TRAIT$ = trait:deviant
			$RITE$ = { rite_has_parameter = deviancy_illegal }
		}
		#INCEST
		AND = {
			$TRAIT$ = trait:incestuous
			NOT = {
				$RITE$ = { rite_has_parameter = allows_unrestricted_marriage }
			}
		}
		#ADULTERER/FORNICATOR
		AND = {
			OR = {
				$TRAIT$ = trait:adulterer
				$TRAIT$ = trait:fornicator
			}
			OR = {
				AND = {
					$RITE$ = { rite_has_parameter = adultery_female_crime }
					$GENDER_CHARACTER$ = { is_female = yes }
				}
				AND = {
					$RITE$ = { rite_has_parameter = adultery_male_crime }
					$GENDER_CHARACTER$ = { is_male = yes }
				}
			}
		}
		# Devoted is criminal when monasticism is forbidden
		AND = {
			$TRAIT$ = trait:devoted
			$RITE$ = { rite_has_parameter = devoted_trait_criminal }
		}
	}
}

relation_with_character_is_incestuous_in_rite_trigger = {
	OR = {
		# doctrine_consanguinity_dynastic; not within the same descent group
		AND = {
			$RITE$ = { rite_has_parameter = consanguinity_restricted_dynastic_marriage }
			is_lowborn = no
			$CHARACTER$.dynasty ?= dynasty
		}
		#doctrine_consanguinity_restricted; absolutely no family business
		AND = {
			$RITE$ = { rite_has_parameter = consanguinity_restricted_close_kin_incest }
			is_close_or_extended_family_of = $CHARACTER$ #[parents, children, siblings, grandparents, grandchildren, cousins, uncles, aunts, nephews, nieces]
		}
		#doctrine_consanguinity_cousins; the only acceptable incest is with your cousin
		AND = {
			$RITE$ = { rite_has_parameter = consanguinity_cousins_marriage }
			OR = {
	            is_close_family_of = $CHARACTER$ #[parents, children, siblings, grandparents, grandchildren]
	            is_uncle_or_aunt_of = $CHARACTER$
	            is_nibling_of = $CHARACTER$
	        }
		}
		#doctrine_consanguinity_aunt_nephew_and_uncle_niece; extended family is ok
		AND = {
			$RITE$ = { rite_has_parameter = consanguinity_aunt_nephew_and_uncle_niece_marriage }
			is_close_family_of = $CHARACTER$
		}
		#doctrine_consanguinity_unrestricted; all forms of incest is acceptable


		#### ME ADDITIONS ####
		AND = {
			exists = $CHARACTER$ # 1.20: vanilla elective triggers can pass an empty character here
			# Saved up front: vanilla passes prev-relative characters (e.g. prev.father), which would resolve differently inside the nested lists below
			$CHARACTER$ = { save_temporary_scope_as = me_affinity_target }
			# Affinity is about your spouse's relatives, never about the spouse: without this every married Muslim counted as
			# committing incest with their own wife or husband (the spouse shares a dynasty with themselves).
			NOT = { is_spouse_of = scope:me_affinity_target }
			OR = {
				### Current Spouse
				# doctrine_me_affinality_polygamy_dynastic; cannot marry anyone from your current spouse's family or dynasty
				AND = {
					OR = {
						$RITE$ = { rite_has_doctrine = doctrine_me_affinality_fixed_dynastic }
						$RITE$ = { rite_has_doctrine = doctrine_me_affinality_polygamy_dynastic }
					}
					OR = {
						any_spouse = {
							OR = {
								is_close_or_extended_family_of = scope:me_affinity_target
								scope:me_affinity_target.dynasty ?= dynasty # only when both have a dynasty
							}
						}
						any_close_or_extended_family_member = { is_spouse_of = scope:me_affinity_target }
						any_dynasty_member = { is_spouse_of = scope:me_affinity_target }
					}
				}
				#doctrine_me_affinality_polygamy_extended_family; cannot marry any of your current spouse's family
				AND = {
					OR = {
						$RITE$ = { rite_has_doctrine = doctrine_me_affinality_fixed_extended_family }
						$RITE$ = { rite_has_doctrine = doctrine_me_affinality_polygamy_extended_family }
					}
					OR = {
						any_spouse = { is_close_or_extended_family_of = scope:me_affinity_target } #[parents, children, siblings, grandparents, grandchildren, cousins, uncles, aunts, nephews, nieces]
						any_close_or_extended_family_member = { is_spouse_of = scope:me_affinity_target }
					}
				}
				#doctrine_me_affinality_polygamy_cousins; cannot marry your current spouse's cousins, siblings, or parents
				AND = {
					OR = {
						$RITE$ = { rite_has_doctrine = doctrine_me_affinality_fixed_cousins }
						$RITE$ = { rite_has_doctrine = doctrine_me_affinality_polygamy_cousins }
					}
					OR = {
						any_spouse = {
							OR = {
								is_close_family_of = scope:me_affinity_target #[parents, children, siblings, grandparents, grandchildren]
								is_cousin_of =  scope:me_affinity_target
							}
						}
						any_close_family_member = { is_spouse_of = scope:me_affinity_target }
						any_close_or_extended_family_member = {
							is_cousin_of_root_trigger = yes
							is_spouse_of = scope:me_affinity_target
						}
					}
				}
				#doctrine_me_affinality_polygamy_close_kin; cannot marry your current spouse's siblings or parents
				AND = {
					OR = {
						$RITE$ = { rite_has_doctrine = doctrine_me_affinality_fixed_close_kin }
						$RITE$ = { rite_has_doctrine = doctrine_me_affinality_polygamy_close_kin }
