### Startup effects
me_add_doctrine_startup_effect = {
	# 1.20: doctrines added to a faith only reach its main Rite, so every other Rite of that faith looked divergent.
	# ME's ablution and affinity doctrines are therefore decided once, for the main Rite, and every other
	# Rite of the faith that has nothing in a group copies the main Rite's doctrine - so they can never differ, and the
	# mod adds no divergence of its own. Rites that define their own doctrine (e.g. the Kapalika) keep it.
	# Safe to run repeatedly; run at game start and yearly (also covers Rites founded during play).
	every_religion_global = {
		every_faith = {
			limit = { exists = main_rite }
			save_temporary_scope_as = me_doctrine_faith
			main_rite = {
				save_temporary_scope_as = me_doctrine_main_rite
				me_main_rite_doctrines_effect = yes
			}
			every_faith_rite = {
				limit = { NOT = { this = scope:me_doctrine_main_rite } }
				me_copy_main_rite_doctrines_effect = yes
			}
		}
	}
}

# Scope: a faith's main Rite; scope:me_doctrine_faith is the faith. Gives it ME's doctrines where it has none.
me_main_rite_doctrines_effect = {
	## Ablution
	if = {
		limit = {
			NOR = {
				rite_has_doctrine = doctrine_me_ablution_none
				rite_has_doctrine = doctrine_me_ablution_handwashing
				rite_has_doctrine = doctrine_me_ablution_full
			}
		}
		if = {
			limit = { scope:me_doctrine_faith = { religion_tag = judaism_religion } }
			add_rite_doctrine = doctrine_me_ablution_handwashing # Jewish handwashing
		}
		else_if = {
			limit = { scope:me_doctrine_faith = { this = faith:mandaean_faith } }
			add_rite_doctrine = doctrine_me_ablution_full # Mandaean rishama
		}
		else_if = {
			limit = { scope:me_doctrine_faith = { NOT = { religion_tag = islam_religion } } }
			add_rite_doctrine = doctrine_me_ablution_none
		}
	}

	## Affinal restrictions
	if = {
		limit = {
			NOT = { has_game_rule = me_affinal_doctrines_no }
			NOR = {
				rite_has_doctrine = doctrine_me_affinality_fixed_dynastic
				rite_has_doctrine = doctrine_me_affinality_fixed_extended_family
				rite_has_doctrine = doctrine_me_affinality_fixed_cousins
				rite_has_doctrine = doctrine_me_affinality_fixed_close_kin
				rite_has_doctrine = doctrine_me_affinality_fixed_parental
				rite_has_doctrine = doctrine_me_affinality_fixed_unrestricted
			}
		}
		if = {
			limit = {
				scope:me_doctrine_faith = {
					OR = {
						is_in_family = rf_abrahamic
						is_in_family = rf_sinitic
					}
				}
				NOT = { rite_has_doctrine = doctrine_consanguinity_unrestricted }
			}
			add_rite_doctrine = doctrine_me_affinality_fixed_parental
		}
		else = {
			add_rite_doctrine = doctrine_me_affinality_fixed_unrestricted
		}
	}
	if = {
		limit = {
			NOT = { has_game_rule = me_affinal_doctrines_no }
			NOR = {
				rite_has_doctrine = doctrine_me_affinality_polygamy_dynastic
				rite_has_doctrine = doctrine_me_affinality_polygamy_extended_family
				rite_has_doctrine = doctrine_me_affinality_polygamy_cousins
				rite_has_doctrine = doctrine_me_affinality_polygamy_close_kin
				rite_has_doctrine = doctrine_me_affinality_polygamy_parental
				rite_has_doctrine = doctrine_me_affinality_polygamy_unrestricted
			}
		}
		if = {
			limit = { scope:me_doctrine_faith = { religion_tag = islam_religion } }
			add_rite_doctrine = doctrine_me_affinality_polygamy_close_kin
		}
		else_if = {
			limit = { rite_has_doctrine = doctrine_me_affinality_fixed_parental }
			add_rite_doctrine = doctrine_me_affinality_polygamy_parental
		}
		else = {
			add_rite_doctrine = doctrine_me_affinality_polygamy_unrestricted
		}
	}
}

				limit = { rite_has_doctrine = doctrine_spiritual_head }

		}
	}

# Scope: an Imam who holds the Imamate. Beyond disease and beyond marriage; his portrait never shows sickbed clothes.
me_imam_protection_effect = {
	if = {
		limit = { NOT = { has_character_flag = immune_to_disease } }
		add_character_flag = immune_to_disease # vanilla can_contract_disease_trait_scope_trigger
	}
	if = {
		limit = { has_character_flag = use_sickness_clothes }
		remove_character_flag = use_sickness_clothes
	}
	if = {
		limit = { NOT = { has_variable = can_not_marry } }
		set_variable = { name = can_not_marry value = yes }
	}
	if = { # what he thinks of anyone is not shown (gui/shared/portraits.gui)
		limit = { NOT = { has_variable = me_hide_opinion } }
		set_variable = { name = me_hide_opinion value = yes }
	}
}

# Scope: none. Gives the Twelver Imamate to the character saved as scope:new_imam.
me_twelver_pass_imamate_effect = {
	if = {
		limit = {
			exists = scope:new_imam
			scope:new_imam = { is_alive = yes }
		}
		if = {
			limit = { NOT = { title:d_imami.holder ?= scope:new_imam } }
			create_title_and_vassal_change = {
				type = returned
				save_scope_as = change
				add_claim_on_loss = no
			}
			title:d_imami = {
				change_title_holder = {
					holder = scope:new_imam
					change = scope:change
				}
			}
			resolve_title_and_vassal_change = scope:change
		}
		scope:new_imam = {
			if = {
				limit = { NOT = { has_character_flag = me_imam_flag } }
				add_character_flag = me_imam_flag
			}
			if = {
				limit = { NOT = { has_government = me_imamate_government } }
				change_government = me_imamate_government
			}
			if = { # the Mahdi has no successor
				limit = {
					has_trait = me_twelver_mahdi
					NOT = { has_realm_law = me_imamate_occultation_law }
				}
				add_realm_law_skip_effects = me_imamate_occultation_law
			}
			else_if = {
				limit = {
					NOT = { has_trait = me_twelver_mahdi }
					NOT = { has_realm_law = me_imamate_succession_law }
				}
				add_realm_law_skip_effects = me_imamate_succession_law
			}
			me_imam_protection_effect = yes
		}
	}
}

	}
	else = {
		create_character = {
			age = { value = current_year subtract = 868 } # born 868
			name = "Muhammad"

			rite = rite:imami
			culture = culture:bedouin

			mother = character:00004447 # Narjis

			trait = intellect_good_3
			trait = education_learning_4
			trait = just
			trait = compassionate
			trait = patient
			trait = saint

		}

# Scope: an Imam (me_imam_flag). The Imam holds no land: whatever reaches him by inheritance, election, a faction or a
# grant goes to the title's de jure liege, failing that to the Imam's own liege, failing that to a new local ruler.
# Duchies and above that nobody can take are destroyed.
me_imam_return_land_effect = {
	if = {
		limit = {
			has_character_flag = me_imam_flag
			any_held_title = {
				tier >= tier_county
				is_landless_type_title = no
				NOT = { this = title:d_imami }
			}
		}
		save_scope_as = me_landed_imam
		every_held_title = {
			limit = {
				tier >= tier_county
				is_landless_type_title = no
				NOT = { this = title:d_imami }
			}
			save_scope_as = me_imam_land
			if = {
				limit = {
					exists = de_jure_liege.holder
					NOT = { de_jure_liege.holder = scope:me_landed_imam }
				}
				de_jure_liege.holder = { save_scope_as = me_imam_land_recipient }
			}
			else_if = {
				limit = {
					exists = de_jure_liege.de_jure_liege.holder
					NOT = { de_jure_liege.de_jure_liege.holder = scope:me_landed_imam }
				}
				de_jure_liege.de_jure_liege.holder = { save_scope_as = me_imam_land_recipient }
			}
			else_if = {
				limit = {
					exists = scope:me_landed_imam.liege
					NOT = { scope:me_landed_imam.liege = scope:me_landed_imam }
				}
				scope:me_landed_imam.liege = { save_scope_as = me_imam_land_recipient }
			}
			else_if = {
				limit = { tier = tier_county }
				create_character = {
					location = scope:me_imam_land.title_province
					culture = scope:me_imam_land.culture
					faith = scope:me_imam_land.faith
					gender = male
					age = { 25 45 }
					save_scope_as = me_imam_land_recipient
				}
			}
			if = {
				limit = { exists = scope:me_imam_land_recipient }
				create_title_and_vassal_change = {
					type = returned
					save_scope_as = change
					add_claim_on_loss = no
				}
				change_title_holder = {
					holder = scope:me_imam_land_recipient
					change = scope:change
				}
				resolve_title_and_vassal_change = scope:change
				clear_saved_scope = me_imam_land_recipient
			}
			else = {
				scope:me_landed_imam = { destroy_title = scope:me_imam_land }
			}
		}
		# back under the Imams' own government
		save_scope_as = new_imam
		me_twelver_pass_imamate_effect = yes
	}
}

### Hijri Calendar Scripted Effects ###

# Initialize the Hijri date based on the game's start date
# Reference dates computed using the tabular Islamic calendar (epoch JDN 1948438):
#   867.1.1   = 17 Dhu al-Hijjah 252 AH
#   1066.9.15 = 16 Shawwal 458 AH
#   1178.1.1  = 3 Rajab 573 AH
#   1337.1.1  = 20 Jumada al-Ula 737 AH
me_initialize_hijri_date_effect = {
	# Only initialize once
	if = {
		limit = {
			NOT = { exists = global_var:me_hijri_year }
		}

		# 867 start date: 17 Dhu al-Hijjah 252 AH
		if = {
			limit = { game_start_date = 867.1.1 }
			set_global_variable = { name = me_hijri_year value = 252 }
			set_global_variable = { name = me_hijri_month value = 12 }
			set_global_variable = { name = me_hijri_day value = 17 }
		}
		# 1066 start date: 16 Shawwal 458 AH
		else_if = {
			limit = { game_start_date = 1066.9.15 }
			set_global_variable = { name = me_hijri_year value = 458 }
			set_global_variable = { name = me_hijri_month value = 10 }
			set_global_variable = { name = me_hijri_day value = 16 }
		}
		# 1178 start date: 3 Rajab 573 AH
		else_if = {
			limit = { game_start_date = 1178.1.1 }
			set_global_variable = { name = me_hijri_year value = 573 }
			set_global_variable = { name = me_hijri_month value = 7 }
			set_global_variable = { name = me_hijri_day value = 3 }
		}
		# 1337 start date: 20 Jumada al-Ula 737 AH
		else_if = {
			limit = { game_start_date = 1337.1.1 }
			set_global_variable = { name = me_hijri_year value = 737 }
			set_global_variable = { name = me_hijri_month value = 5 }
			set_global_variable = { name = me_hijri_day value = 20 }
		}
		# Fallback for any other start date: approximate from formula
		else = {
			set_global_variable = {
				name = me_hijri_year
				value = me_hijri_year_from_formula
			}
			set_global_variable = { name = me_hijri_month value = 1 }
			set_global_variable = { name = me_hijri_day value = 1 }
		}
	}
}

# Advance the Hijri calendar by exactly 1 day (called daily via on_action)
# Hijri month lengths:
#   Odd months (1,3,5,7,9,11): 30 days
#   Even months (2,4,6,8,10): 29 days
#   Month 12: 29 days normally, 30 days in leap years
#   Leap years in 30-year cycle: 2, 5, 7, 10, 13, 16, 18, 21, 24, 26, 29
me_advance_hijri_day_effect = {
	if = {
		limit = { exists = global_var:me_hijri_year }

		# Increment day by 1
		change_global_variable = { name = me_hijri_day add = 1 }

		# Check overflow: odd month (30 days), day hit 31
		if = {
			limit = {
				me_hijri_month_is_odd >= 1
				me_hijri_day_display >= 31
			}
			set_global_variable = { name = me_hijri_day value = 1 }
			change_global_variable = { name = me_hijri_month add = 1 }
		}
		# Check overflow: month 12 in leap year (30 days), day hit 31
		else_if = {
			limit = {
				me_hijri_month_value >= 12
				me_hijri_month_value < 13
				me_hijri_leap_check < 11
				me_hijri_day_display >= 31
			}
			set_global_variable = { name = me_hijri_day value = 1 }
			change_global_variable = { name = me_hijri_month add = 1 }
		}
		# Check overflow: even month (29 days), day hit 30
		else_if = {
			limit = {
				me_hijri_month_is_odd < 1
				me_hijri_day_display >= 30
			}
			set_global_variable = { name = me_hijri_day value = 1 }
			change_global_variable = { name = me_hijri_month add = 1 }
		}

		# Year rollover: month exceeded 12
		if = {
			limit = { me_hijri_month_value >= 13 }
			set_global_variable = { name = me_hijri_month value = 1 }
			change_global_variable = { name = me_hijri_year add = 1 }
		}
	}
}

# — called from the Hijri daily pulse on the month-start boundary, and from the
# decisions themselves if the holy month happens to already be in progress.
#
# Modifier durations:
#   Ramadan fasting/patronage: 30 days (covers the full lunar month)
#   Muharram mourning/patronage: 12 days (Ashura + surrounding mourning)
#   Tatbir: 10 days (through Ashura itself)

### Apply personal Muharram mourning modifier based on stored approach ###
me_apply_muharram_modifier_effect = {
	if = {
		limit = { exists = var:me_muharram_approach }
		if = { limit = { has_character_modifier = me_muharram_deep_mourning_modifier } remove_character_modifier = me_muharram_deep_mourning_modifier }
		if = { limit = { has_character_modifier = me_muharram_moderate_mourning_modifier } remove_character_modifier = me_muharram_moderate_mourning_modifier }
		if = { limit = { has_character_modifier = me_muharram_tatbir_modifier } remove_character_modifier = me_muharram_tatbir_modifier }
		if = { limit = { has_character_modifier = me_muharram_skipped_modifier } remove_character_modifier = me_muharram_skipped_modifier }
		# Deep
		if = {
			limit = { var:me_muharram_approach = flag:deep }
			add_character_modifier = {
				modifier = me_muharram_deep_mourning_modifier
				days = 12
			}
			stress_impact = {
				impatient = medium_stress_impact_gain
				cynical = medium_stress_impact_gain
				callous = minor_stress_impact_gain
				zealous = minor_stress_impact_loss
				compassionate = minor_stress_impact_loss
			}
		}
		# Moderate
		else_if = {
			limit = { var:me_muharram_approach = flag:moderate }
			add_character_modifier = {
				modifier = me_muharram_moderate_mourning_modifier
				days = 12
			}
			stress_impact = {
				zealous = minor_stress_impact_gain
			}
		}
		# Tatbir
		else_if = {
			limit = { var:me_muharram_approach = flag:tatbir }
			add_character_modifier = {
				modifier = me_muharram_tatbir_modifier
				days = 10
			}
			stress_impact = {
				cynical = medium_stress_impact_gain
				craven = minor_stress_impact_gain
				flagellant = medium_stress_impact_loss
				brave = minor_stress_impact_loss
			}
		}
		# Skip
		else_if = {
			limit = { var:me_muharram_approach = flag:skip }
			add_character_modifier = {
				modifier = me_muharram_skipped_modifier
				days = 12
			}
			stress_impact = {
				zealous = major_stress_impact_gain
				compassionate = minor_stress_impact_gain
				cynical = minor_stress_impact_loss
			}
		}
	}
}

ME_create_artifact_book_effect = {
	$OWNER$ = { save_scope_as = owner } #Will always exist!
	$CREATOR$ = { #Doesn't always exist
		save_scope_as = author
	}
	save_temporary_scope_value_as = {
		name = subject_is_set
		value = $SET_SUBJECT$
	}
	save_temporary_scope_value_as = {
		name = topic_is_set
		value = $SET_TOPIC$
	}
	if = {
		limit = {
			exists = scope:dummy_gender
			scope:author = scope:dummy_gender
		}
		clear_saved_scope = author
		clear_saved_scope = dummy_gender
	}
	else = { #If we have an author we save them as a variable on the owner so that we can access them later on when setting the subject etc.
		scope:owner = {
			set_variable = {
				name = book_artifact_author
				value = scope:author
				days = 1
			}
		}
	}
	hidden_effect_new_object = {
		# Get artifact quality and wealth
		get_artifact_quality_effect = yes
		get_artifact_wealth_effect = yes
		

		# To save the subject, topic, and other variables we'll need if it has been set in events or in the effect
		get_book_variables_effect = yes

		if = { #Should this be a court artifact?
			limit = {
				NOT = { exists = scope:should_be_trinket }
			}
			if = { #Do we have an author?
				limit = {
					exists = scope:author
				}
				if = { #Is this author using Palm Leaf (indian) books?
					limit = {
						scope:author = { artifact_use_indian_building_visuals_trigger = yes }
					}
					scope:owner = {
						create_artifact = {	
							name = placeholder # Will be regenerated in the post-creation effects
							creator = scope:author
							description = placeholder # Will be regenerated in the post-creation effects
							visuals = indian_book
							type = book
							modifier = artifact_placeholder_modifier
							save_scope_as = newly_created_artifact
							wealth = scope:wealth
							quality = scope:quality
							#decaying = no
						}
					}
				}
				else = { #Codices/Non-Indian books
					scope:owner = {
						create_artifact = {	
							name = placeholder # Will be regenerated in the post-creation effects
							creator = scope:author
							description = placeholder # Will be regenerated in the post-creation effects
							visuals = book
							type = book
							modifier = artifact_placeholder_modifier
							save_scope_as = newly_created_artifact
							wealth = scope:wealth
							quality = scope:quality
							#decaying = no
						}
					}
				}
			}
			else = { #We don't have an author
				if = {
					limit = { #Is this a palm leaf/indian book?
						scope:owner = { artifact_use_indian_building_visuals_trigger = yes }
					}
					scope:owner = {
						create_artifact = {	
							name = placeholder # Will be regenerated in the post-creation effects
							description = placeholder # Will be regenerated in the post-creation effects
							visuals = indian_book
							type = book
							modifier = artifact_placeholder_modifier
							save_scope_as = newly_created_artifact
							wealth = scope:wealth
							quality = scope:quality
							#decaying = no
						}
					}
				}
				else = { #This is a codex/non-indian book:
					scope:owner = {
						create_artifact = {	
							name = placeholder # Will be regenerated in the post-creation effects
							description = placeholder # Will be regenerated in the post-creation effects
							visuals = book
							type = book
							history = { type = created_before_history } # We don't know who wrote it
							modifier = artifact_placeholder_modifier
							save_scope_as = newly_created_artifact
							wealth = scope:wealth
							quality = scope:quality
							#decaying = no
						}
					}
				}
			}
		}
		else = { #This is a court artifact
			if = {
				limit = {
					exists = scope:author #We have an author
				}
				if = {
					limit = {
						scope:author = { artifact_use_indian_building_visuals_trigger = yes }
					}
					scope:owner = {
						create_artifact = {	
							name = placeholder # Will be regenerated in the post-creation effects
							creator = scope:author
							description = placeholder # Will be regenerated in the post-creation effects
							visuals = indian_book
							type = miscellaneous
							modifier = artifact_placeholder_modifier
							save_scope_as = newly_created_artifact
							wealth = scope:wealth
							quality = scope:quality
						}
					}
					# Subject and cover material is set as a required feature for a book-type artifact, but for a miscellaneous-type it's an optional feature, so we need to set it here
					scope:newly_created_artifact = {
						set_artifact_feature_group = book_subject
						set_artifact_feature_group = book_cover_material
					}
				}
				else = { #This is a codex/non-indian book
					scope:owner = {
						create_artifact = {	
							name = placeholder # Will be regenerated in the post-creation effects
							creator = scope:author
							description = placeholder # Will be regenerated in the post-creation effects
							visuals = book
							type = miscellaneous
							modifier = artifact_placeholder_modifier
							save_scope_as = newly_created_artifact
							wealth = scope:wealth
							quality = scope:quality
							
						}
					}
					# Subject and cover material is set as a required feature for a book-type artifact, but for a miscellaneous-type it's an optional feature, so we need to set it here
					scope:newly_created_artifact = {
						set_artifact_feature_group = book_subject
						set_artifact_feature_group = book_cover_material
					}
				}
			}
			else = { #We don't have an author
				if = {
					limit = { #Is this a palm-leaf/indian book?
						scope:owner = { artifact_use_indian_building_visuals_trigger = yes }
					}
					scope:owner = {
						create_artifact = {	
							name = placeholder # Will be regenerated in the post-creation effects
							description = placeholder # Will be regenerated in the post-creation effects
							visuals = indian_book
							type = miscellaneous
							modifier = artifact_placeholder_modifier
							save_scope_as = newly_created_artifact
							wealth = scope:wealth
							quality = scope:quality
						}
					}
					# Subject and cover material is set as a required feature for a book-type artifact, but for a miscellaneous-type it's an optional feature, so we need to set it here
					scope:newly_created_artifact = {
						set_artifact_feature_group = book_subject
						set_artifact_feature_group = book_cover_material
					}
				}
				else = { #This isn't a palm leaf/indian book
					scope:owner = {
						create_artifact = {	
							name = placeholder # Will be regenerated in the post-creation effects
							description = placeholder # Will be regenerated in the post-creation effects
							visuals = book
							type = miscellaneous
							history = { type = created_before_history } # We don't know who wrote it
							modifier = artifact_placeholder_modifier
							save_scope_as = newly_created_artifact
							wealth = scope:wealth
							quality = scope:quality
						}
					}
					# Subject and cover material is set as a required feature for a book-type artifact, but for a miscellaneous-type it's an optional feature, so we need to 	set it here
					scope:newly_created_artifact = {
						set_artifact_feature_group = book_subject
						set_artifact_feature_group = book_cover_material
					}
				}
			}
		}

		scope:newly_created_artifact = {
			set_artifact_feature_group = book_topic
			set_artifact_feature_group = book_title
			set_artifact_feature_group = book_decoration
			set_artifact_feature_group = decoration_pattern
			get_artifact_feature_references_effect = yes
			# Get quality of written content
			if = {
				limit = {
					NOT = { exists = scope:author }
				}
				get_book_content_quality_effect = { SKILL_BASE = scope:owner }
			}
			else = {
				get_book_content_quality_effect = { SKILL_BASE = scope:author }
			}
			# We save these values as variables on the owner so that we can access them deep in scopes in the desc-loc later on
			scope:owner = {
				set_variable = {
					name = quality
					value = scope:quality
				}
				set_variable = {
					name = wealth
					value = scope:wealth
				}
				set_variable = {
					name = book_content_quality
					value = scope:book_content_quality
				}
			}
			# Generate the updated - final description for the book
			set_artifact_description = artifact_book_description
			# Generate the updated title - final title for the book
			set_artifact_name = artifact_book_title
			if = {
				limit = {
					has_artifact_feature = book_subject_al_ghazali_revival
				}
				add_artifact_modifier = artifact_monthly_piety_1_modifier # 0.1
				add_artifact_modifier = artifact_learning_per_piety_level_modifier
				remove_artifact_modifier = artifact_placeholder_modifier
			}
			else_if = {
				limit = {
					has_artifact_feature = book_subject_al_ghazali_deliverance
				}
				add_artifact_modifier = artifact_monthly_piety_1_modifier # 0.1
				add_artifact_modifier = artifact_stress_gain_2_modifier
				remove_artifact_modifier = artifact_placeholder_modifier
			}
			# Modifiers applied based on topic
			#
			#
			#

			set_variable = {
				name = quality
				value = scope:quality
			}
			set_variable = {
				name = wealth
				value = scope:wealth
			}
			set_variable = {
				name = book_content_quality
				value = scope:book_content_quality
			}
		}
	}

	#To clean out variables
	if = {
		limit = {
			exists = scope:author
		}
		scope:author = {
			if = {
				limit = {
					has_variable = book_subject
				}
				remove_variable = book_subject
			}
			if = {
				limit = {
					has_variable = book_topic
				}
				remove_variable = book_topic
			}
			if = {
				limit = {
					has_variable = book_topic_category
				}
				remove_variable = book_topic_category
			}
			if = {
				limit = {
					has_variable = book_content_quality
				}
				remove_variable = book_content_quality
			}
		}
	}
	scope:owner = {
		if = {
			limit = { has_variable = book_artifact_subject }
			remove_variable = book_artifact_subject
		}
		if = {
			limit = { has_variable = book_artifact_topic }
			remove_variable = book_artifact_topic
		}
		if = {
			limit = { has_variable = book_topic_category }
			remove_variable = book_topic_category
		}
		if = {
			limit = {
				has_variable = book_topic_category
			}
			remove_variable = book_topic_category
		}
		if = {
			limit = {
				has_variable = book_content_quality
			}
			remove_variable = book_content_quality
		}
	}

	if = {
		limit = {
			exists = scope:should_be_trinket 
		}
		clear_saved_scope = should_be_trinket 	
	}
}
