appoint_a_righteous_caliph_scripted_effect = {
	# Create a HoF title & assign it appropriately.
	hidden_effect = {

		### ME Addition: the Ibadi and Wahbi Imamates (vanilla helper, which also sets scope:hof_assigned) ###
		give_or_make_temporal_head_of_faith_title = { RITE = rite:ibadi TITLE = title:d_ibadi }
		give_or_make_temporal_head_of_faith_title = { RITE = rite:wahbiya TITLE = title:d_wahbiya }
		give_or_make_temporal_head_of_faith_title = { RITE = rite:maliki TITLE = title:d_muwalladi }
		give_or_make_temporal_head_of_faith_title = { RITE = rite:nizari TITLE = title:d_nizari }
		give_or_make_temporal_head_of_faith_title = { RITE = rite:zayidi TITLE = title:d_zayidi }
		give_or_make_temporal_head_of_faith_title = { RITE = rite:ismaili TITLE = title:d_ismaili }
		give_or_make_temporal_head_of_faith_title = { RITE = rite:imami TITLE = title:d_imami }

		# If not, follow the usual flow.
		if = {
			limit = { NOT = { exists = scope:hof_assigned } }
			set_up_dynamic_temporal_hof_title_effect = { NEW_HOLDER = scope:founder }
			faith = { set_religious_head_title = scope:new_title }
		}
	}
	custom_tooltip = appoint_a_righteous_caliph_decision.tt.you_become_caliph
	# Give scope:founder an extra piety level, if possible.
	if = {
		limit = { piety_level < max_piety_level }
		add_piety_level = 1
	}
	# Try to scope:founder rivals with scope:old_caliph.
	if = {
		limit = {
			can_set_relation_rival_trigger = { CHARACTER = scope:old_caliph }
		}
		set_relation_rival_effect = {
			TARGET = scope:old_caliph
			REASON = rival_old_caliph_new_caliph
		}
	}
	# If you've pulled a fast one on your HoF, they see through it.
	if = {
		limit = { has_character_flag = wishes_to_schism_caliphate_flag }
		scope:old_caliph = {
			remove_opinion = {
				target = scope:founder
				modifier = repentant_opinion
			}
		}
		remove_hook = {
			target = scope:old_caliph
			type = indebted_hook
		}
	}
	# And sort a bit of stress impact for the gravity of things.
	stress_and_fulfillment_impact = {
		base = major_stress_impact_gain
		humble = major_stress_impact_gain
		arrogant = medium_stress_impact_loss
	}
}

# ME: vanilla 1.21 effect, overridden so the +30 opinion for founding a Military Order goes to the Head of Faith the founder
# follows (his rival Head of Faith if he backs one) and not to a Head of Faith the founder himself rivals.
create_holy_order_accompanying_effect = {
	hidden_effect = {
		if = { # set up currencies for the HO leader, has to be saved as scope:leader
			limit = {
				exists = scope:leader
			}
			scope:leader = {
				add_gold = 100 #So that they have some money to lend out
				add_piety_level = 2
				add_gold = holy_order_starting_gold
				#Get rid of non-martial gender courtiers before they get armored up
				every_courtier = {
					limit = {
						NOT = {
							can_be_combatant_based_on_gender_trigger = { ARMY_OWNER = scope:leader }
						}
					}
					scope:leader = {
						remove_courtier_or_guest = prev
					}
				}
				every_courtier = {
					add_trait = order_member
				}
			}

			# send messages
			send_interface_toast = {
				type = holy_order_founded_message
				desc = i_created_holy_order_message
				left_icon = scope:leader
				right_icon = scope:new_holy_order.title
			}
			if = {
				limit = {
					has_personal_tenet_flag = leasing_to_military_order_give_spiritual_fulfillment_personal
					faith = scope:leader.faith
				}
				send_interface_toast = {
					type = event_generic_good
					title = leasing_to_military_order_give_spiritual_fulfillment_personal_toast
					change_spiritual_fulfillment = medium_spiritual_fulfillment_value
				}
			}
		}
		# send messages to relevant characters that HO got created
		save_scope_as = holy_order_creator
		every_player = {
			limit = {
				faith = scope:holy_order_creator.faith
				this != scope:holy_order_creator
			}
			send_interface_message = {
				type = holy_order_founded_message
				desc = someone_created_holy_order_message
				left_icon = scope:holy_order_creator
				right_icon = scope:new_holy_order.title
			}
		}
		every_neighboring_top_liege_realm_owner = {
			limit = {
				faith != scope:holy_order_creator.faith
			}
			send_interface_message = {
				type = enemy_holy_order_founded_message
				desc = other_faith_neighbor_created_holy_order_message
				left_icon = scope:holy_order_creator
				right_icon = scope:new_holy_order.title
			}
		}
		# get the achievements for creating a HO
		add_achievement_flag_effect = { FLAG = achievement_non_nobis_domine_flag }
		if = {
			limit = { exists = global_var:started_give_a_dog_a_bone_achievement }
			add_achievement_global_variable_effect = {
				VARIABLE = holy_order_give_a_dog_a_bone_achievement
				VALUE = yes
			}
		}
	}
	# gain opinion with your HoF, if you have one and it's not you
	if = {
		limit = {
			exists = religious_head_or_challenger
			religious_head_or_challenger != root
		}
		religious_head_or_challenger = {
			add_opinion = {
				target = root
				modifier = founded_holy_order_opinion
			}
		}
	}
	# get a nice modifier
	if = {
		limit = {
			NOT = { has_character_modifier = founded_holy_order_modifier }
		}
		add_character_modifier = {
			modifier = founded_holy_order_modifier
			years = 5
		}
	}
	# and boost the fervor
	faith = {
		change_fervor = {
			value = significant_fervor_gain
			desc = fervor_gain_holy_order_founded
		}
	}
}
