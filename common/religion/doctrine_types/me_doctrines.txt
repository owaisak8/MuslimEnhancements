		value = faith_doctrine_cost_low

		}

			limit = {
				OR = {
					rite_has_doctrine = islam_secondary_group_sunni_heresy_doctrine

			text = must_select_sunni
			doctrine:muhammad_succession_sunni_doctrine = { is_in_list = selected_doctrines }
		}
	}

		hostility_level_override = {
			islam_secondary_group_sunni_orthodox_doctrine = righteous
			islam_secondary_group_sunni_heresy_doctrine = astray
			islam_secondary_group_shia_batriya_doctrine = astray

	doctrine_group_type = doctrine_me_ablution
	index = 0
	divergence = 0
	can_pick = {
		custom_description = { # Only Zandaqa can refuse to do Wudhu
			text = must_select_zandaqa
			OR = {
				NOT = { religion_tag = islam_religion }
				AND = {
					religion_tag = islam_religion
					doctrine:muhammad_succession_zandaqa_doctrine = { is_in_list = selected_doctrines }
				}
			}
		}
	}
	piety_cost = {
		value = faith_doctrine_cost_low
		if = {
			limit = { rite_has_doctrine = doctrine_me_ablution_handwashing }
			multiply = faith_unchanged_doctrine_cost_mult
		}
		else_if = {
			limit = { rite_has_doctrine = doctrine_me_ablution_full }
			multiply = faith_unchanged_doctrine_cost_mult
		}
	}
	parameters = {
		me_ablution_none
	}
}

## Affinal Restrictions

	doctrine_group_type = doctrine_me_affinality_fixed
	index = 0
	divergence = 0
	piety_cost = {
		value = faith_doctrine_cost_low
		if = {
			limit = { rite_has_doctrine = doctrine_me_affinality_fixed_dynastic }
			multiply = faith_unchanged_doctrine_cost_mult
		}
		else_if = {
			limit = { rite_has_doctrine = doctrine_me_affinality_fixed_extended_family }
			multiply = faith_changed_doctrine_cost_mult_two_step
		}
		else_if = {
			limit = { rite_has_doctrine = doctrine_me_affinality_fixed_cousins }
			multiply = faith_changed_doctrine_cost_mult_two_step
		}
		else_if = {
			limit = { rite_has_doctrine = doctrine_me_affinality_fixed_close_kin }
			multiply = faith_changed_doctrine_cost_mult_three_step
		}
		else_if = {
			limit = { rite_has_doctrine = doctrine_me_affinality_fixed_parental }
			multiply = faith_changed_doctrine_cost_mult_three_step
		}
		else_if = {
			limit = { rite_has_doctrine = doctrine_me_affinality_fixed_unrestricted }
			multiply = faith_changed_doctrine_cost_mult_four_step
		}
	}
	
	is_shown = { NOT = { has_game_rule = me_affinal_doctrines_no } }
	
	can_pick = {
		NOT = { has_game_rule = me_affinal_doctrines_no }
		custom_description = {
			text = incompatible_doctrine_me_affinal_consanguinity_trigger
			NOR = {
				doctrine:doctrine_consanguinity_restricted = { is_in_list = selected_doctrines }
				doctrine:doctrine_consanguinity_cousins = { is_in_list = selected_doctrines }
				doctrine:doctrine_consanguinity_aunt_nephew_and_uncle_niece = { is_in_list = selected_doctrines }
				doctrine:doctrine_consanguinity_unrestricted = { is_in_list = selected_doctrines }
			}
		}
	}

	parameters = {
		affinal_restricted_dynastic_marriage_fixed
		affinal_restricted_extended_kin_marriage_fixed
		#affinal_restricted_extended_kin_incest_fixed = yes
	}
}
