		controller = create_holy_order
		title_valid = {
			trigger_if = {
				limit = { exists = this }
				NOR = {
					this.county = { has_county_modifier = me_iftar_sponsorship_modest_modifier }
					this.county = { has_county_modifier = me_iftar_sponsorship_substantial_modifier }
					this.county = { has_county_modifier = me_iftar_sponsorship_grand_modifier }
				}
				OR = {
					AND = {
						this.county.rite = { rite_has_doctrine = doctrine_me_fasting_month_ramadan }
						me_hijri_month_value = 9
					}
					AND = {
						this.county.rite = { rite_has_doctrine = doctrine_me_fasting_month_rajab }
						me_hijri_month_value = 7
					}
				}
				is_capital_barony = yes
				is_leased_out = no
