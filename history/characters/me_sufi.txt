### Sufi masters and other Friends of God (Awliya) ###
# Shaykhs of the Tariqas (see common/scripted_effects/me_tariqa_effects.txt) and revered figures whose tombs are places of visitation.

# A dated effect after a death marks the tomb as that of a Wali for games starting later (the same way vanilla presets its saints).

	name = "Junayd"
	dynasty = me_sufi_baghdadi

		birth = yes
	}
	910.1.1 = {
		death = yes
	}
	910.2.1 = {
		effect_even_if_dead = {
			if = {
				# From 1166 the Baghdad tomb site represents Abd al-Qadir al-Jilani (a province holds one saint)
				limit = { game_start_date < 1166.2.1 }
				pam_canonize_saint_effect = {
					CANONIZER = root
					CANDIDATE = root
					CONSECRATION_PROVINCE = province:4828 # Baghdad
					PRESET = yes
				}
			}
			else = { # ME: buried in the Shuniziyya cemetery, Baghdad; registered at Aiwana, the nearest free province
				pam_canonize_saint_effect = {
					CANONIZER = root
					CANDIDATE = root
					CONSECRATION_PROVINCE = province:4832
					PRESET = yes
				}
			}
		}
	}
}
