# ME: flavour events for the Muslim world

# 0002 - A madrasa is founded (me_found_madrasa_decision)

# Khutba and sikka
me_flavour.0001 = {
	type = character_event
	title = me_flavour.0001.t
	desc = me_flavour.0001.desc
	theme = realm
	left_portrait = {
		character = root
		animation = personality_honorable
	}
	right_portrait = {
		trigger = { exists = scope:me_hof }
		character = scope:me_hof
	}

	immediate = {
		faith.religious_head ?= {
			if = {
				limit = { NOT = { this = root } }
				save_scope_as = me_hof
			}
		}
		add_character_modifier = me_sikka_modifier
	}

	# The Head of Faith first, then me
	option = {
		name = me_flavour.0001.a
		trigger = { exists = scope:me_hof }
		scope:me_hof = {
			add_opinion = {
				target = root
				modifier = me_named_in_khutba_opinion
			}
		}
		add_piety = medium_piety_gain
		add_prestige = minor_prestige_gain
		ai_chance = {
			base = 100
		}
	}

	# My name alone
	option = {
		name = me_flavour.0001.b
		add_prestige = major_prestige_gain
		if = {
			limit = { exists = scope:me_hof }
			scope:me_hof = {
				add_opinion = {
					target = root
					modifier = me_left_out_of_khutba_opinion
				}
			}
		}
		stress_impact = {
			humble = medium_stress_impact_gain
		}
		ai_chance = {
			base = 25
			modifier = {
				add = 50
				has_trait = ambitious
			}
		}
	}
}
